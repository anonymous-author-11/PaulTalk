
def yielding_fn(n : i32) -> i32 {
	print(n);
	yield();
	n = n + 1;
	print(n);
	yield();
	n = n + 1;
	print(n);
	yield();
	n = n + 1;
	print(n);
	return n;
}

def helper() {
	yield();
}

def yielding_callee_in_loop(k : i32) {
	helper();

	while k < 10 {
		helper();
		k = k + 1;
	}

	helper();
}

def passthru_fn(n : i32) -> i32 {
	return yielding_fn(n : i32);
}

def calling_fn(n : i32) {
	coro = Coro{passthru_fn, n};
	coro.call();
	n = n + 10;
	print(n);
	coro.call();
	n = n + 20;
	print(n);
	return;
}

def main() {
	calling_fn(5);
}
